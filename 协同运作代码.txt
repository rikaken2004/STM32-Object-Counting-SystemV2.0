#include <reg52.h>
typedef unsigned char uchar;
typedef unsigned int uint;

// 端口定义
sbit LED1 = P3^6;
sbit LED2 = P3^7;

// 串口初始化（9600bps，11.0592MHz晶振）
void UART_Init(void) {
    TMOD |= 0x20;  // 定时器1工作模式2（8位自动重装）
    TH1 = 0xFD;    // 9600bps初值
    TL1 = 0xFD;
    TR1 = 1;       // 启动定时器1
    SCON = 0x50;   // 串口工作模式1（8位UART，允许接收）
    EA = 1;        // 开启总中断
    ES = 1;        // 开启串口中断
}

// 串口发送一个字节
void UART_SendByte(uchar dat) {
    SBUF = dat;
    while(!TI);    // 等待发送完成
    TI = 0;        // 清除发送标志位
}

// 延时函数（ms级，11.0592MHz晶振）
void Delay_ms(uint ms) {
    uint i, j;
    for(i = ms; i > 0; i--)
        for(j = 110; j > 0; j--);
}

// 主机控制逻辑：LED1/LED2闪烁3次后发送指令
void Main(void) {
    uchar cnt = 0;
    UART_Init();   // 初始化串口
    while(1) {
        // 闪烁3次（亮500ms+灭500ms为1次）
        while(cnt < 3) {
            LED1 = 0;  // 亮
            LED2 = 0;
            Delay_ms(500);
            LED1 = 1;  // 灭
            LED2 = 1;
            Delay_ms(500);
            cnt++;
        }
        UART_SendByte(0x01);  // 发送触发指令
        while(1);  // 发送后停止闪烁，等待从机响应（可扩展其他逻辑）
    }
}

// 串口中断服务函数（可选：接收从机应答）
void UART_ISR(void) interrupt 4 {
    uchar rec_dat;
    if(RI) {       // 接收中断标志
        rec_dat = SBUF;
        RI = 0;     // 清除接收标志位
        // 可添加应答处理逻辑（如收到从机确认后做其他操作）
    }
}

#include <reg51.h>
#include <intrins.h>

// 从单片机P2口控制L298N的引脚定义
sbit ENA   = P2^0;  // L298N使能A
sbit ENB   = P2^1;  // L298N使能B
sbit IN1   = P2^2;  // 电机A转向
sbit IN2   = P2^3;
sbit IN3   = P2^4;  // 电机B转向
sbit IN4   = P2^5;

// 变量定义
bit stop_flag = 0;      // 停止指令标志
unsigned int delay_time = 5000; // 5秒延迟重启

// 精准毫秒延时（11.0592MHz晶振）
void DelayMs(unsigned int ms)
{
    unsigned int i,j;
    for(i=ms; i>0; i--)
        for(j=110; j>0; j--); // 固定110，确保延时精准
}

// 串口初始化
void UART_Init(void)
{
    TMOD = 0x20;       // 定时器1方式2（单独赋值，避免位操作冲突）
    TH1 = 0xFD;        
    TL1 = 0xFD;
    TR1 = 1;           
    SCON = 0x50;       
    EA = 1;            // 开总中断
    ES = 1;            // 开串口中断
    TI = 0;            // 手动清发送标志
    RI = 0;            // 手动清接收标志
}

// 串口发送函数
void UART_Send(unsigned char dat)
{
    SBUF = dat;
    while(TI == 0);  
    TI = 0;          
}

// 【关键修复】简化串口中断，确保stop_flag必置位
void UART_ISR(void) interrupt 4
{
    unsigned char recv_data;
    if(RI) // 仅处理接收中断
    {
        recv_data = SBUF; // 读取指令
        RI = 0;           // 清接收标志
        if(recv_data == 0x01)
        {
            stop_flag = 1; // 强制置位停止标志
        }
    }
}

// 【关键修复】电机启停逻辑：增加LED提示（便于调试）
sbit LED_DEBUG = P1^0; // 新增调试LED，电机停止时亮
void Motor_Stop_Restart(void)
{
    if(stop_flag == 1)
    {
        LED_DEBUG = 1;   // 调试LED亮，提示电机停止
        ENA = 0;         // 停止电机A
        ENB = 0;         // 停止电机B
        DelayMs(delay_time);  // 延迟5秒
        ENA = 1;         // 重启电机A
        ENB = 1;         // 重启电机B
        LED_DEBUG = 0;   // 调试LED灭，提示电机重启
        UART_Send(0x02); // 必发回传指令
        stop_flag = 0;   // 清标志
    }
}

void main(void)
{
    UART_Init();  
    // 电机转向初始化
    IN1 = 0; IN2 = 1;
    IN3 = 0; IN4 = 1;
    // 电机默认启动
    ENA = 1;
    ENB = 1;
    LED_DEBUG = 0; // 调试LED初始灭

    while(1)
    {
        Motor_Stop_Restart();  
    }
}
